ferramentas.py python 
import datetime
import math

# ── 1. Definição das ferramentas (o que o Claude vê) ──────────────
FERRAMENTAS = [
    {
        "name": "calcular",
        "description": "Calcula expressões matemáticas. Use para qualquer conta.",
        "input_schema": {
            "type": "object",
            "properties": {
                "expressao": {
                    "type": "string",
                    "description": "Expressão matemática. Ex: '2 ** 10' ou 'math.sqrt(144)'"
                }
            },
            "required": ["expressao"]
        }
    },
    {
        "name": "data_hora",
        "description": "Retorna a data e hora atual do sistema.",
        "input_schema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    },
    {
        "name": "contar_palavras",
        "description": "Conta quantas palavras há em um texto.",
        "input_schema": {
            "type": "object",
            "properties": {
                "texto": {
                    "type": "string",
                    "description": "Texto a ser analisado"
                }
            },
            "required": ["texto"]
        }
    }
]

# ── 2. Implementação real das ferramentas (funções Python) ─────────
def executar_ferramenta(nome: str, params: dict) -> str:
    """Executa a ferramenta pedida pelo Claude e retorna o resultado."""

    if nome == "calcular":
        try:
            # Avalia a expressão de forma segura
            resultado = eval(params["expressao"], {"__builtins__": {}}, {"math": math})
            return str(resultado)
        except Exception as e:
            return f"Erro: {e}"

    elif nome == "data_hora":
        agora = datetime.datetime.now()
        return agora.strftime("%d/%m/%Y %H:%M:%S")

    elif nome == "contar_palavras":
        palavras = params["texto"].split()
        return f"{len(palavras)} palavras"

    return "Ferramenta desconhecida."

O Loop do Agente agent loop

Agentic Loop Tool Result stop_reason Iteração
O coração do agente é um loop: enviamos a mensagem ao Claude, verificamos se ele quer usar uma ferramenta (stop_reason == "tool_use"), executamos a ferramenta, devolvemos o resultado, e repetimos — até o Claude responder com "end_turn", indicando que terminou.
💡 O Claude não executa ferramentas por conta própria — ele pede para você executar. Isso é intencional: mantém o controle do que é executado do seu lado, não dentro do modelo.
agente.py — loop principal python 
from ferramentas import FERRAMENTAS, executar_ferramenta

def rodar_agente(pergunta: str) -> str:
    """
    Roda o loop completo do agente até obter a resposta final.
    Retorna o texto final gerado pelo Claude.
    """
    print(f"\n🧠 Pergunta: {pergunta}")
    print("─" * 50)

    # Histórico de mensagens (memória de curto prazo)
    mensagens = [
        {"role": "user", "content": pergunta}
    ]

    # Limite de iterações para evitar loop infinito
    max_iteracoes = 10
    iteracao = 0

    while iteracao < max_iteracoes:
        iteracao += 1
        print(f"\n🔄 Iteração {iteracao}...")

        # ── Envia mensagens ao Claude ──────────────────────────────
        resposta = cliente.messages.create(
            model="claude-sonnet-4-5",
            max_tokens=1024,
            tools=FERRAMENTAS,          # lista de ferramentas disponíveis
            messages=mensagens           # histórico completo
        )

        # ── Adiciona resposta do Claude ao histórico ───────────────
        mensagens.append({
            "role": "assistant",
            "content": resposta.content
        })

        # ── Verificar motivo de parada ─────────────────────────────
        if resposta.stop_reason == "end_turn":
            # Claude terminou — extrai texto final
            for bloco in resposta.content:
                if hasattr(bloco, "text"):
                    print(f"\n✅ Resposta final:\n{bloco.text}")
                    return bloco.text

        elif resposta.stop_reason == "tool_use":
            # Claude quer usar ferramenta(s)
            resultados_tools = []

            for bloco in resposta.content:
                if bloco.type == "tool_use":
                    print(f"  🔧 Usando ferramenta: {bloco.name}")
                    print(f"     Parâmetros: {bloco.input}")

                    # Executa a ferramenta e captura resultado
                    resultado = executar_ferramenta(bloco.name, bloco.input)
                    print(f"     Resultado: {resultado}")

                    resultados_tools.append({
                        "type": "tool_result",
                        "tool_use_id": bloco.id,   # ID único da chamada
                        "content": resultado
                    })

            # Devolve os resultados ao Claude para continuar
            mensagens.append({
                "role": "user",
                "content": resultados_tools
            })

    return "Limite de iterações atingido."


# ── Teste ────────────────────────────────────────────────────────
if __name__ == "__main__":
    rodar_agente("Quanto é 2 elevado a 20? E qual é a data de hoje?")

Adicionando Memória memory

Short-term Memory Histórico Context Window
O agente acima esquece tudo entre chamadas a rodar_agente(). Para criar uma sessão com memória, precisamos manter o histórico de mensagens fora da função e passá-lo de uma chamada à outra — simulando uma conversa contínua.
agente_com_memoria.py python 
class AgenteComMemoria:
    """Agente que mantém histórico entre perguntas."""

    def __init__(self, system_prompt: str = ""):
        self.historico = []     # memória de curto prazo
        self.system_prompt = system_prompt

    def perguntar(self, pergunta: str) -> str:
        # Adiciona nova pergunta ao histórico
        self.historico.append({"role": "user", "content": pergunta})

        max_iter = 10
        for _ in range(max_iter):
            kwargs = {
                "model": "claude-sonnet-4-5",
                "max_tokens": 1024,
                "tools": FERRAMENTAS,
                "messages": self.historico
            }
            # System prompt é opcional (instrui o comportamento do agente)
            if self.system_prompt:
                kwargs["system"] = self.system_prompt

            resposta = cliente.messages.create(**kwargs)
            self.historico.append({
                "role": "assistant", "content": resposta.content
            })

            if resposta.stop_reason == "end_turn":
                for b in resposta.content:
                    if hasattr(b, "text"):
                        return b.text

            elif resposta.stop_reason == "tool_use":
                tool_results = []
                for b in resposta.content:
                    if b.type == "tool_use":
                        res = executar_ferramenta(b.name, b.input)
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": b.id,
                            "content": res
                        })
                self.historico.append({
                    "role": "user", "content": tool_results
                })

        return "Limite de iterações."

    def limpar_memoria(self):
        """Reseta a conversa."""
        self.historico = []
        print("🗑️  Memória apagada.")


# ── Teste de conversa contínua ────────────────────────────────────
if __name__ == "__main__":
    agente = AgenteComMemoria(
        system_prompt="Você é um assistente matemático. Seja direto e use as ferramentas disponíveis."
    )

    print(agente.perguntar("Quanto é 15 ao quadrado?"))
    print(agente.perguntar("Agora some esse resultado com 50."))  # lembra o resultado anterior!
    print(agente.perguntar("E qual a raiz quadrada disso?"))

  • LangChain / LangGraph — frameworks de alto nível para construir pipelines agentic com grafos de estado
